### 3.子任务三：数据表的创建及维护
(1)`use house_market;`
```mysql
create table house_info (

```
(2)
```mysql
create table community_info(
	community_id int,
	community_name varchar(100),
	district varchar(50),
	build_year year,
	property_fee decimal(10,2),
	subway_distance int,
	primary key(community_id)
);
```
(3)
```mysql
create table transaction_records(
	record_id int,
	house_id int,
	transaction_date date,
	transaction_price decimal(12,2),
	price_change decimal(12,2),
	transaction_type varchar(50),
	primary key(record_id)
);
```
(4)统计每个区域的在售房源数量和平均价格
```mysql
select
	district,
	count(*),
	avg(price) as 平均价格
from
	house_info
where
	status = '在售'
group by
	 district;
```
查询成交价格高于该区域平均成交价格的房源信息
```mysql
select
	h.*,
	t.*
from
	house_info as h
join
	transaction_records as t
on
	h.house_id = t.house_id
where
	t.transaction_price > (
	select
		avg(t2.transaction_price)
	from
		transaction_records as t2
	join
		house_info as h2
	on
		t2.house_id = h2.house_id
	where h2.district = h.district
	);
```
查询距离地铁站1000米以内的小区及其房源数量
```mysql
select
	c.community_name,
	count(h.house_id) as 房源数量
from
	house_info as h
join
	community_info as c
on
	c.community_name = h.community
where
	c.subway_distance < 1000
group by
	c.community_name;
```
统计每种户型的平均单价（按面积计算）并按降序排列
```mysql
select
	h.layout
	avg(h.price/h.area) as 平均单价
from
	house_info as h
where
	h.area > 0
group by
	h.layout
order by
	平均单价 desc;
```
查询最近一个月内成交的房源信息及其所在小区详情
```mysql
select
	t.*,
	h.*,
	c.*
from
	transaction_records as t
join
	house_info as h
on
	t.house_id = h.house_id
join
	community_info as c
on
	h.community = c.community_name
where
	t.transaction_date >=subdate(curdate(),interval 1 month);
```
