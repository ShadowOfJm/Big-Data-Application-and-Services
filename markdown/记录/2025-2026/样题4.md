(1)
```mysql
create table destination(
	dest_id int primary key auto_increment comment '目的地ID',
```

(2)
```mysql
create table travel_record(
	record_id bigint primary key auto_increment comment '记录ID',
	user_id varchar(50) comment '用户ID，脱敏处理',
	dest_id int comment '目的地ID，外键，关联 destination 表',
	travel_date date comment '出行日期',
	travel_days int comment '出行天数',
	total_expense decimal(10,2) comment '总花费（元）',
	transport_mode varchar(20) comment '主要交通方式，如：飞机、高铁、自驾',
	is_group_tour tinyint(1) comment '是否跟团游 1 是，0 否',
	satisfaction int comment '满意度评分 1-5 分',
	foreign key (dest_id) references destination(dest_id)
);
```
(3)
```mysql
select quarter(travel_date) as 季度,
       count(*) as 出行总人次,
       round(sum(total_expense)/count(*),2) as 人均花费
from travel_record
where year(travel_date) = 2024
group by quarter(travel_date)
order by 季度;
```

```mysql
select d.dest_type as 目的地类型,
       sum(case when r.is_group_tour = 1 then 1 else 0 end) as 跟团游次数,
       sum(case when r.is_group_tour = 0 then 1 else 0 end) as 自由行次数,
       round(sum(case when r.is_group_tour = 1 then 1 else 0 end)/count(*)*100,1) as 跟团游占比
from travel_record r
join destination d on r.dest_id = d.dest_id
group by d.dest_type
order by 跟团游次数 desc;
```

```mysql
select d.dest_name as 目的地名称,
       d.country as 所在国家,
       d.estimated_cost as 预估人均费用,
       round(sum(r.total_expense)/count(*),2) as 实际人均花费,
       round(sum(r.total_expense)/count(*) - d.estimated_cost,2) as 差值
from travel_record r
join destination d on r.dest_id = d.dest_id
group by d.dest_id, d.dest_name, d.country, d.estimated_cost
having 差值 > d.estimated_cost * 0.2
order by 差值 desc;
```
